% Slide 2: Research Overview
\begin{frame}{Research Vision}
    \begin{itemize}
        \item \textbf{Core Objective:} Bridging the gap between raw energy data and actionable knowledge using Semantic Web technologies.
        \item \textbf{Key Pillars:}
            \begin{itemize}
                \item Knowledge Representation (Ontologies/Knowledge Graphs).
                \item Information Extraction from heterogeneous sources.
                \item Implementation of FAIR Principles (Findable, Accessible, Interoperable, Reusable).
                \item Advancement of Open Science in the Energy Domain.
            \end{itemize}
    \end{itemize}
\end{frame}

% Slide 3: The Energy Domain Context
\begin{frame}{Information Technology for Energy Data Management (IT4EDM)}
    \begin{itemize}
        \item Located at the \textbf{IAI (KIT)}, focusing on the digital transformation of energy systems.
        \item \textbf{Challenges:} Data silos, inconsistent metadata, and high-frequency time-series data.
        \item \textbf{Our Approach:} Developing robust IT methods to streamline data discovery and integration for researchers and grid operators.
    \end{itemize}
\end{frame}

% --- Slide 4: ReGIMO (Part 1: The Challenge & Concept) ---
\begin{frame}{ReGIMO: Bridging the Metadata Gap}
    \begin{itemize}
        \item \textbf{The Challenge:} 
            \begin{itemize}
                \item High manual effort in metadata enrichment.
                \item Heterogeneous definitions across different energy domains.
                \item Obstacles to FAIR data workflows due to "siloed" descriptions.
            \end{itemize}
        \item \textbf{Our Solution:} An automated tool for the \textbf{registration of metadata} on distributed infrastructures.
        \item \textbf{Core Philosophy:} Enhancing findability and machine-readability \textbf{without} altering data ownership or moving the actual data.
    \end{itemize}
    \vfill
    \tiny{Source: Koubaa et al. (2024). \textit{Automatic Metadata Registration using regimo}. Zenodo. DOI: 10.5281/zenodo.10671031}
\end{frame}

% --- Slide 5: ReGIMO (Part 2: Technical Features) ---
\begin{frame}{ReGIMO: Technical Implementation}
    \begin{itemize}
        \item \textbf{Semantic Interoperability:} 
            \begin{itemize}
                \item Maps local structures to standardized schemas.
                \item Integration of domain-specific vocabularies.
            \end{itemize}
        \item \textbf{Workflow Integration:}
            \begin{itemize}
                \item Seamless connection with \textbf{RDMO} (Research Data Management Organiser).
                \item Reuses existing project metadata to minimize redundant entry.
            \end{itemize}
        \item \textbf{Deployment:} Target infrastructures include the \textbf{Open Energy Family Databus}, fostering regionalized information modeling.
    \end{itemize}
\end{frame}

% --- Slide 6: Semantico (Part 1: Knowledge Extraction) ---
\begin{frame}{Semantico: From Strings to Semantics}
    \begin{itemize}
        \item \textbf{The Challenge:} The "Literature Deluge"—e.g., the rapid growth of Perovskite solar cell research makes manual review impossible.
        \item \textbf{Objective:} A semi-automatic toolchain to find, analyze, and classify literature using \textbf{Ontology Learning}.
        \item \textbf{Methodology:} 
            \begin{itemize}
                \item Collaboration between IT systems and domain experts.
                \item Defining "Representational Artifacts" (Taxonomies/Relations).
                \item Automated identification of concepts and universals in text.
            \end{itemize}
    \end{itemize}
    \vfill
    \tiny{Source: Koubaa et al. (2023). \textit{Semantico – Toolchain for literature classification}. Zenodo. DOI: 10.5281/zenodo.8421250}
\end{frame}

% --- Slide 7: Semantico (Part 2: Applications) ---
\begin{frame}{Semantico: Impact on Energy Research}
    \begin{itemize}
        \item \textbf{Semantic Mapping:}
            \begin{itemize}
                \item Automated classification into domain classes.
                \item Linking extracted data to global Knowledge Graphs.
            \end{itemize}
        \item \textbf{FAIR Data Enabling:} 
            \begin{itemize}
                \item Turns unstructured PDF data into machine-actionable knowledge.

            \end{itemize}
        \item \textbf{Utility:} Supports meta-analyses and rapid data discovery for experimental researchers.
    \end{itemize}
\end{frame}

% Slide 9: Open Science & Information Extraction
\begin{frame}{The Path to Open Science}
    \begin{itemize}
        \item \textbf{Information Extraction:} Using NLP and heuristic methods to "unlock" data trapped in PDFs and legacy databases.
        \item \textbf{Open Science:} Commitment to open-source software and open-access data publications (Zenodo).
